import { DatabaseSync } from 'node:sqlite';
import { readFileSync } from 'node:fs';
import { fileURLToPath } from 'node:url';
import { dirname, join } from 'node:path';

const __filename = fileURLToPath(import.meta.url);
const __dirname = dirname(__filename);

const dbPath = join(__dirname, 'database.sqlite');
export const db = new DatabaseSync(dbPath);

// Initialize schema on load
export function initDatabase() {
  const schemaPath = join(__dirname, 'schema.sql');
  const schemaSql = readFileSync(schemaPath, 'utf8');
  db.exec(schemaSql);
  console.log('[DB] SQLite Database schema initialized successfully at:', dbPath);
}

// Helper wrapper functions
export const query = {
  all(sql, ...params) {
    const stmt = db.prepare(sql);
    return stmt.all(...params);
  },
  get(sql, ...params) {
    const stmt = db.prepare(sql);
    return stmt.get(...params);
  },
  run(sql, ...params) {
    const stmt = db.prepare(sql);
    return stmt.run(...params);
  },
  exec(sql) {
    return db.exec(sql);
  }
};

export default db;

import jwt from 'jsonwebtoken';
import { query } from '../db/index.js';

export const JWT_SECRET = process.env.JWT_SECRET || 'universe_diu_super_secure_jwt_secret_2026';

export function verifyToken(req, res, next) {
  const authHeader = req.headers['authorization'];
  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    return res.status(401).json({ error: 'Authentication required. Missing or invalid Bearer token.' });
  }

  const token = authHeader.split(' ')[1];
  try {
    const decoded = jwt.verify(token, JWT_SECRET);
    // Fetch fresh user record
    const user = query.get('SELECT id, name, email, student_id, role, department, batch, avatar_url, knowledge_points, status FROM users WHERE id = ?', decoded.id);
    
    if (!user) {
      return res.status(401).json({ error: 'User no longer exists.' });
    }
    if (user.status === 'SUSPENDED' || user.status === 'BANNED') {
      return res.status(403).json({ error: 'Your account has been suspended. Please contact the administrator.' });
    }

    req.user = user;
    next();
  } catch (err) {
    return res.status(401).json({ error: 'Invalid or expired session token.' });
  }
}

// Optional Auth (for public feed requests that can still show if user has liked/saved)
export function optionalAuth(req, res, next) {
  const authHeader = req.headers['authorization'];
  if (authHeader && authHeader.startsWith('Bearer ')) {
    const token = authHeader.split(' ')[1];
    try {
      const decoded = jwt.verify(token, JWT_SECRET);
      const user = query.get('SELECT id, name, email, role, status FROM users WHERE id = ?', decoded.id);
      if (user && user.status === 'ACTIVE') {
        req.user = user;
      }
    } catch (e) {
      // Ignore token error for optional auth
    }
  }
  next();
}

// Role-based Access Control Guard
export function requireRole(allowedRoles) {
  return (req, res, next) => {
    if (!req.user) {
      return res.status(401).json({ error: 'Authentication required.' });
    }
    if (!allowedRoles.includes(req.user.role)) {
      return res.status(403).json({ 
        error: `Access denied. Requires one of [${allowedRoles.join(', ')}] privileges. Your role is ${req.user.role}.` 
      });
    }
    next();
  };
}

import { Router } from 'express';
import bcrypt from 'bcryptjs';
import jwt from 'jsonwebtoken';
import { query } from '../db/index.js';
import { JWT_SECRET, verifyToken } from '../middleware/auth.js';

const router = Router();

// POST /api/auth/login
router.post('/login', (req, res) => {
  const { emailOrStudentId, password } = req.body;

  if (!emailOrStudentId || !password) {
    return res.status(400).json({ error: 'Please provide student ID or email, and password.' });
  }

  // Lookup by email or student ID
  const user = query.get(
    'SELECT * FROM users WHERE email = ? OR student_id = ?',
    emailOrStudentId.trim().toLowerCase(),
    emailOrStudentId.trim()
  );

  if (!user) {
    return res.status(401).json({ error: 'Invalid university credentials.' });
  }

  if (user.status === 'SUSPENDED') {
    return res.status(403).json({ error: 'Your account is suspended. Contact the DIU administrator.' });
  }

  const isMatch = bcrypt.compareSync(password, user.password_hash);
  if (!isMatch) {
    return res.status(401).json({ error: 'Incorrect password.' });
  }

  const token = jwt.sign(
    { id: user.id, email: user.email, role: user.role, studentId: user.student_id },
    JWT_SECRET,
    { expiresIn: '7d' }
  );

  const { password_hash, ...sanitizedUser } = user;
  return res.json({
    message: 'Login successful',
    token,
    user: sanitizedUser
  });
});

// POST /api/auth/register
router.post('/register', (req, res) => {
  const { name, email, studentId, password, department, batch } = req.body;

  if (!name || !email || !studentId || !password) {
    return res.status(400).json({ error: 'All fields (Name, DIU Email, Student ID, Password) are required.' });
  }

  const cleanEmail = email.trim().toLowerCase();
  if (!cleanEmail.endsWith('@diu.edu.bd')) {
    return res.status(400).json({ error: 'Registration is restricted strictly to official DIU emails (@diu.edu.bd).' });
  }

  // Check unique constraints
  const existing = query.get('SELECT id FROM users WHERE email = ? OR student_id = ?', cleanEmail, studentId.trim());
  if (existing) {
    return res.status(409).json({ error: 'A student account with this email or Student ID already exists.' });
  }

  const hash = bcrypt.hashSync(password, 10);
  const defaultAvatar = `https://images.unsplash.com/photo-1534528741775-53994a69daeb?w=120&auto=format&fit=crop&q=80`;

  const result = query.run(`
    INSERT INTO users (name, email, student_id, password_hash, role, department, batch, avatar_url, knowledge_points)
    VALUES (?, ?, ?, ?, 'STUDENT', ?, ?, ?, 10)
  `, name.trim(), cleanEmail, studentId.trim(), hash, department || 'Software Engineering', batch || '43', defaultAvatar);

  const newUser = query.get('SELECT id, name, email, student_id, role, department, batch, avatar_url, knowledge_points, status FROM users WHERE id = ?', result.lastInsertRowid);

  const token = jwt.sign(
    { id: newUser.id, email: newUser.email, role: newUser.role, studentId: newUser.student_id },
    JWT_SECRET,
    { expiresIn: '7d' }
  );

  return res.status(201).json({
    message: 'Account created successfully! Welcome to UniverseDIU.',
    token,
    user: newUser
  });
});

// GET /api/auth/me
router.get('/me', verifyToken, (req, res) => {
  // Aggregate user stats
  const postsCount = query.get('SELECT count(*) as count FROM posts WHERE author_id = ?', req.user.id).count;
  const resourcesCount = query.get("SELECT count(*) as count FROM posts WHERE author_id = ? AND type = 'Resource'", req.user.id).count;
  const answersCount = query.get('SELECT count(*) as count FROM comments WHERE author_id = ?', req.user.id).count;
  const helpfulCount = query.get('SELECT COALESCE(SUM(helpful_count), 0) as total FROM posts WHERE author_id = ?', req.user.id).total;

  res.json({
    user: {
      ...req.user,
      stats: {
        posts: postsCount,
        resources: resourcesCount,
        answers: answersCount,
        helpful: helpfulCount
      }
    }
  });
});

export default router;

      p.id as post_id, p.title as post_title, p.content as post_content,
      u.name as reporter_name, u.email as reporter_email
    FROM reports r

  `;

  if (!report) {
    return res.status(404).json({ error: 'Report not found.' });

  query.run('UPDATE reports SET status = ? WHERE id = ?', status || 'RESOLVED', reportId);

  const { course, type, search, limit = 50 } = req.query;
  const currentUserId = req.user ? req.user.id : 0;

  let sql = `
    SELECT 
      p.id, p.type, p.title, p.content, p.tags, p.helpful_count, p.comments_count, p.created_at,
      u.id as author_id, u.name as author_name, u.department as author_department, u.batch as author_batch, u.avatar_url as author_avatar,

      EXISTS(SELECT 1 FROM saved_posts WHERE post_id = p.id AND user_id = ?) as is_saved
    FROM posts p
    JOIN users u ON p.author_id = u.id
    LEFT JOIN courses c ON p.course_id = c.id
    LEFT JOIN attachments a ON a.post_id = p.id
    WHERE p.status = 'PUBLISHED'
  `;

    params.push(type);

router.post('/', verifyToken, (req, res) => {
  const { title, content, type = 'Knowledge', courseCode = 'SE-322', tags = [], attachment } = req.body;

    return res.status(400).json({ error: 'Post title and description are required.' });

  }

  const insertPost = query.run(`
    INSERT INTO posts (author_id, course_id, type, title, content, tags)
    VALUES (?, ?, ?, ?, ?, ?)
  `, req.user.id, courseId, type, title.trim(), content.trim(), tagsJson);

  const postId = insertPost.lastInsertRowid;

  // Insert attachment if provided
  if (attachment && attachment.name) {
    query.run(`
      INSERT INTO attachments (post_id, file_name, file_size, file_type, file_url)
      VALUES (?, ?, ?, ?, ?)
    `, postId, attachment.name, attachment.size || '1.0 MB', attachment.type || 'PDF', attachment.url || '');
  }

  // Award +10 Knowledge points to author
  query.run('UPDATE users SET knowledge_points = knowledge_points + 10 WHERE id = ?', req.user.id);

  res.status(201).json({
    message: 'Post published successfully!',
    postId
  });
});

// POST /api/posts/:id/helpful - Toggle Helpful Upvote
router.post('/:id/helpful', verifyToken, (req, res) => {
  const postId = req.params.id;
  const userId = req.user.id;

  const existing = query.get('SELECT 1 FROM post_likes WHERE user_id = ? AND post_id = ?', userId, postId);

  if (existing) {
    // Unlike
    query.run('DELETE FROM post_likes WHERE user_id = ? AND post_id = ?', userId, postId);
    query.run('UPDATE posts SET helpful_count = MAX(0, helpful_count - 1) WHERE id = ?', postId);
    // Deduct points
    query.run('UPDATE users SET knowledge_points = MAX(0, knowledge_points - 5) WHERE id = (SELECT author_id FROM posts WHERE id = ?)', postId);
    const updated = query.get('SELECT helpful_count FROM posts WHERE id = ?', postId);
    return res.json({ hasLiked: false, helpfulCount: updated.helpful_count });
  } else {
    // Like
    query.run('INSERT INTO post_likes (user_id, post_id) VALUES (?, ?)', userId, postId);
    query.run('UPDATE posts SET helpful_count = helpful_count + 1 WHERE id = ?', postId);
    // Award +5 points to author
    query.run('UPDATE users SET knowledge_points = knowledge_points + 5 WHERE id = (SELECT author_id FROM posts WHERE id = ?)', postId);
    const updated = query.get('SELECT helpful_count FROM posts WHERE id = ?', postId);
    return res.json({ hasLiked: true, helpfulCount: updated.helpful_count });
  }

import { Router } from 'express';

const router = Router();

// GET /api/comments/post/:postId - Get Threaded Comments
router.get('/post/:postId', (req, res) => {
  const postId = req.params.postId;

  const sql = `
    SELECT 
      c.id, c.content, c.parent_comment_id, c.helpful_count, c.created_at,
      u.id as author_id, u.name as author_name, u.department as author_department, 
      u.batch as author_batch, u.avatar_url as author_avatar, u.role as author_role,
      (p.author_id = u.id) as is_post_author
    FROM comments c
    JOIN users u ON c.author_id = u.id
    JOIN posts p ON c.post_id = p.id
    WHERE c.post_id = ?
    ORDER BY c.created_at ASC
  `;

  const rows = query.all(sql, postId);

  // Separate root comments and nested replies
  const rootComments = [];
  const repliesMap = {};

  rows.forEach(r => {
    const comment = {
      id: r.id,
      content: r.content,
      helpfulCount: r.helpful_count,
      createdAt: r.created_at,
      author: {
        id: r.author_id,
        name: r.author_name,
        meta: `${r.author_department} • Batch ${r.author_batch}`,
        avatar: r.author_avatar,
        role: r.author_role,
        isAuthor: Boolean(r.is_post_author)
      },
      replies: []
    };

    if (r.parent_comment_id) {
      if (!repliesMap[r.parent_comment_id]) {
        repliesMap[r.parent_comment_id] = [];
      }
      repliesMap[r.parent_comment_id].push(comment);
    } else {
      rootComments.push(comment);
    }
  });

  // Attach replies
  rootComments.forEach(root => {
    if (repliesMap[root.id]) {
      root.replies = repliesMap[root.id];
    }
  });

  res.json({ comments: rootComments });
});

-- UniverseDIU MySQL Database Schema
-- MySQL 8.0+ | InnoDB | utf8mb4

CREATE TABLE IF NOT EXISTS users (
  id INT NOT NULL AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL,
  student_id VARCHAR(100) NOT NULL,
  password_hash VARCHAR(255) NOT NULL,
  role ENUM('STUDENT','MODERATOR','ADMIN','SUPER_ADMIN') DEFAULT 'STUDENT',
  department VARCHAR(255) DEFAULT 'Software Engineering',
  batch VARCHAR(50) DEFAULT '43',
  avatar_url TEXT,
  knowledge_points INT DEFAULT 0,
  status ENUM('ACTIVE','SUSPENDED','BANNED') DEFAULT 'ACTIVE',
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uq_users_email (email),
  UNIQUE KEY uq_users_student_id (student_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS departments (
  id INT NOT NULL AUTO_INCREMENT,
  code VARCHAR(50) NOT NULL,
  name VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY uq_departments_code (code)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS courses (
  id INT NOT NULL AUTO_INCREMENT,
  code VARCHAR(50) NOT NULL,
  name VARCHAR(255) NOT NULL,
  department_id INT DEFAULT NULL,
  semester VARCHAR(100) DEFAULT 'Spring 2026',
  instructor_name VARCHAR(255) DEFAULT NULL,
  instructor_title VARCHAR(255) DEFAULT NULL,
  description TEXT,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uq_courses_code (code),
  CONSTRAINT fk_courses_department
    FOREIGN KEY (department_id) REFERENCES departments(id)
    
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS posts (
  id INT NOT NULL AUTO_INCREMENT,
  author_id INT NOT NULL,
  course_id INT DEFAULT NULL,
  type ENUM('Knowledge','Resource','Question','Project','Link','Media') DEFAULT 'Knowledge',
  title VARCHAR(500) NOT NULL,
  content LONGTEXT NOT NULL,
  tags LONGTEXT NOT NULL,
  helpful_count INT DEFAULT 0,
  comments_count INT DEFAULT 0,
  status ENUM('PUBLISHED','HIDDEN','REMOVED') DEFAULT 'PUBLISHED',
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_posts_author (author_id),
  KEY idx_posts_course (course_id),
  KEY idx_posts_status_created (status, created_at),
  CONSTRAINT fk_posts_author
    FOREIGN KEY (author_id) REFERENCES users(id)
    ON DELETE CASCADE,
  CONSTRAINT fk_posts_course
    FOREIGN KEY (course_id) REFERENCES courses(id)
    
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS attachments (
  id INT NOT NULL AUTO_INCREMENT,
  post_id INT NOT NULL,
  file_name VARCHAR(255) NOT NULL,
  file_size VARCHAR(100) DEFAULT NULL,
  file_type VARCHAR(100) DEFAULT NULL,
  file_url TEXT,
  PRIMARY KEY (id),
  KEY idx_attachments_post (post_id),
  CONSTRAINT fk_attachments_post
    FOREIGN KEY (post_id) REFERENCES posts(id)
    ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS comments (
  id INT NOT NULL AUTO_INCREMENT,
  post_id INT NOT NULL,
  author_id INT NOT NULL,
  parent_comment_id INT DEFAULT NULL,
  content TEXT NOT NULL,
  helpful_count INT DEFAULT 0,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_comments_post (post_id),
  KEY idx_comments_author (author_id),
  KEY idx_comments_parent (parent_comment_id),
  CONSTRAINT fk_comments_post
    FOREIGN KEY (post_id) REFERENCES posts(id)
    ON DELETE CASCADE,
  CONSTRAINT fk_comments_author
    FOREIGN KEY (author_id) REFERENCES users(id)
    ON DELETE CASCADE,
  CONSTRAINT fk_comments_parent
    FOREIGN KEY (parent_comment_id) REFERENCES comments(id)
    ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

    FOREIGN KEY (post_id) REFERENCES posts(id)
    ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS saved_posts (
  user_id INT NOT NULL,
  post_id INT NOT NULL,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (user_id, post_id),
  KEY idx_saved_posts_post (post_id),
  CONSTRAINT fk_saved_posts_user
    FOREIGN KEY (user_id) REFERENCES users(id)
    ON DELETE CASCADE,
  CONSTRAINT fk_saved_posts_post
    FOREIGN KEY (post_id) REFERENCES posts(id)
    ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS reports (
  id INT NOT NULL AUTO_INCREMENT,
  post_id INT NOT NULL,
  reporter_id INT NOT NULL,
  reason TEXT NOT NULL,
  status ENUM('PENDING','RESOLVED','DISMISSED') DEFAULT 'PENDING',
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_reports_post (post_id),
  KEY idx_reports_reporter (reporter_id),
  KEY idx_reports_status (status),
  CONSTRAINT fk_reports_post
    FOREIGN KEY (post_id) REFERENCES posts(id)
    ON DELETE CASCADE,
  CONSTRAINT fk_reports_reporter
    FOREIGN KEY (reporter_id) REFERENCES users(id)
    ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS audit_logs (
  id INT NOT NULL AUTO_INCREMENT,
  actor_id INT NOT NULL,
  action VARCHAR(255) NOT NULL,
  target_type VARCHAR(100) NOT NULL,
  target_id INT NOT NULL,
  details TEXT,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_audit_actor (actor_id),
  KEY idx_audit_created (created_at),
  CONSTRAINT fk_audit_actor
    FOREIGN KEY (actor_id) REFERENCES users(id)
    ON DELETE RESTRICT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
